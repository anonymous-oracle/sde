                         The (Self) Attention Mechanism
o Attention as a "fuzzy" or approximate hashtable
    o   To look up a value, we compare a query against keys in a table
    o   In a hashtable (shown on the bottom left):
         o   Each query (hash) maps to exactly one key-value pair
    o   In (self-)attention (shown on the bottom right)
         o   Each query matches each key to varying degrees.
         o   We return a sum of values weighted by the query-key match.

def positional_encoding(max_len, d_model):
    pe = torch.zeros(max_len, d_model)                                        Deterministic
    pos = torch.arange(max_len).unsqueeze(1).float()    # (max_len, 1)
    div = torch.exp(torch.arange(0, d_model, 2).float()                   No training needed for positions
                    * -(math.log(10000.0) / d_model))

    pe[:, 0::2] = torch.sin(pos * div)   # even dims: sine
                                                                              Unique per position
    pe[:, 1::2] = torch.cos(pos * div)   # odd dims: cosine
    return pe   # shape: (max_len, d_model)                               Each pos has a unique fingerprint

class Transformer(nn.Module):
    def __init__(self, vocab_size=50000, d_model=512,
                 n_heads=8, n_layers=6, d_ff=2048, max_len=512):
        super().__init__()
        self.embedding = TransformerEmbedding(vocab_size, d_model, max_len)
        self.encoder   = nn.ModuleList(
            [TransformerEncoderLayer(d_model, n_heads, d_ff) for _ in range(n_layers)])
        self.fc_out    = nn.Linear(d_model, vocab_size)

    def forward(self, src):
        x = self.embedding(src)          # token + position
        for layer in self.encoder:
            x = layer(x)                 # 6× self-attention + FFN
        return self.fc_out(x)            # project to vocab

     Start Token                                         Forward Pass

     Feed <BOS> as the first decoder input.              Masked Self-Attn -> Cross-Attn -> FFN.



     Linear + Softmax                                    Sample / Argmax

def generate(model, src, max_len=50, bos=1, eos=2):
    enc_out = model.encoder(src)               # encode source once
    tgt = torch.tensor([[bos]])                    # start with <BOS>
    for _ in range(max_len):
          mask = causal_mask(tgt.size(1))       # rebuild mask each step
          logits = model.decoder(tgt, enc_out, mask)[:, -1, :]
          next_tok = logits.argmax(dim=-1).unsqueeze(0)        # greedy decode
          tgt = torch.cat([tgt, next_tok], dim=-1)      # append token
          if next_tok.item() == eos: break      # stop at <EOS>
    return tgt
